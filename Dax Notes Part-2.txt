11. CAGR for EV States = 
-- To ensure that the context of each State is respected when calculating the first and last year sales, we are calculating the minimum and maximum fiscal year within the context of each maker.
VAR FirstYear = CALCULATE(MIN(dates[fiscal_year]), ALL(dates))
VAR LastYear = CALCULATE(MAX(dates[fiscal_year]), ALL(dates))

VAR FirstYearEVSales = 
    CALCULATE(
        [EV_Sold_State],
       dates[fiscal_year] = FirstYear
    )

VAR LastYearEVSales = 
    CALCULATE(
        [EV_Sold_State],
        dates[fiscal_year] = LastYear
    )

VAR NumberOfYears = LastYear - FirstYear

RETURN
IF(
    NumberOfYears > 0,
    ( DIVIDE(LastYearEVSales, FirstYearEVSales, 0) ^ DIVIDE( 1, NumberOfYears, 0 ) ) - 1,
    BLANK()
)

15. Ranking_States_Penetration = 
VAR _penetrationThreshold = 0
VAR _filteredStates = 
    FILTER(
        ALL(dim_states[state]), 
        [Penetration_Rate] > _penetrationThreshold
    )
// The _filteredStates variable filters out any states with Penetration Rate less than or equal to _penetrationThreshold, which is set to 0. This ensures only states with a penetration rate greater than 0 are considered.
VAR _topStates = 
    RANKX(
        _filteredStates, 
        [Penetration_Rate], 
        , 
        DESC, 
        DENSE
    )
VAR _bottomStates = 
    RANKX(
        _filteredStates, 
        [Penetration_Rate], 
        , 
        ASC, 
        DENSE
    )
VAR _ranking = 
    SWITCH(
        TRUE(),
        SELECTEDVALUE('TopBottom'[Value]) = "Top", _topStates,
        SELECTEDVALUE('TopBottom'[Value]) = "Bottom", _bottomStates
    )
RETURN
IF(
    AND(
        _ranking <= SELECTEDVALUE('TopN Slicers for States'[TopN States]), 
        [Penetration_Rate] > _penetrationThreshold
    ), 
    [Penetration_Rate]
)
// The final RETURN statement only returns the penetration rate for states whose rank is within the selected TopN range and whose penetration rate is greater than the threshold.

24. Overall Decline Relative Chg = 
// Absolute Change: simpler and provides a straightforward comparison in percentage points.
// [PR 2024] - [PR 2022]

// Relative Change: (with division) provides context on how significant the change is relative to the initial value, which can be more insightful in certain analyses.
VAR PR22 = [PR 2022]
VAR PR24 = [PR 2024]
RETURN
IF(
    ISBLANK(PR22) && ISBLANK(PR24),  -- If both are BLANK (filtered out), return BLANK
    BLANK(),
    IF(
        PR22 > 0 || PR24 > 0,  -- Consider states where there's data in either year
        DIVIDE(PR24 - PR22, PR22, 0),  -- Calculate relative change
        BLANK()
    )
)

-- Checking if there is an overall decline in penetration rate over the period:
IF(
    ISBLANK(PR22) && ISBLANK(PR23) && ISBLANK(PR24),
    BLANK(),
    IF(
        PR24 < PR23 && PR23 < PR22 && PR22 > 0,  -- Decline only if PR22 > 0 and subsequent years show a decrease
        "Is Declining YoY",
        "Not Declining YoY"
    )
)
